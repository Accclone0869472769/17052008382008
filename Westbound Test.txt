local SAVE_FOLDER = "StarCrowHub"
local SAVE_FILE   = SAVE_FOLDER .. "/WestBoundConfig.json"

local function EnsureFolder()
    if not isfolder(SAVE_FOLDER) then makefolder(SAVE_FOLDER) end
end

local function SaveConfig()
    pcall(function()
        EnsureFolder()
        local data = {}
        for k, v in pairs(S.Config) do data[k] = v end
        writefile(SAVE_FILE, HttpService:JSONEncode(data))
    end)
end

local function LoadConfig()
    pcall(function()
        if isfile(SAVE_FILE) then
            local raw = readfile(SAVE_FILE)
            local data = HttpService:JSONDecode(raw)
            for k, v in pairs(data) do
                if S.Config[k] ~= nil then S.Config[k] = v end
            end
        end
    end)
end

local LastCacheUpdate = 0

local function GetRobTargets()
    local now = os.clock()
    if now - LastCacheUpdate > 5 then
        LastCacheUpdate = now
        table.clear(RobTargetsCache)
        for _, obj in ipairs(workspace:GetDescendants()) do
            local name = obj.Name
            if name == "Safe" or name == "Register" or name == "Cash" or name == "Gold" or name == "Loot" then
                local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart", true)
                if part then
                    table.insert(RobTargetsCache, { obj = obj, part = part, isSafe = (name == "Safe") })
                end
            end
        end
    end
    return RobTargetsCache
end

local function CreateFloatButton(name, text, defaultPos, configKey)
    local btn = Instance.new("TextButton")
    btn.Name = name
    btn.Size = UDim2.new(0, S.Config.FloatButtonSize, 0, S.Config.FloatButtonSize)
    
    local savedPos = S.Config[configKey]
    if type(savedPos) == "table" and #savedPos == 4 then
        btn.Position = UDim2.new(savedPos[1], savedPos[2], savedPos[3], savedPos[4])
    else
        btn.Position = defaultPos
    end
    
    btn.BackgroundColor3 = Color3.fromRGB(62, 62, 62)
    btn.BackgroundTransparency = 1
    btn.AutoButtonColor = false
    btn.Text = text
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 13
    btn.Visible = false
    btn.Parent = FloatGui

    Instance.new("UICorner", btn).CornerRadius = UDim.new(0.25, 0)
    local stroke = Instance.new("UIStroke", btn)
    stroke.Thickness = 1.6
    stroke.Color = Color3.fromRGB(255, 255, 255)
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border

    local grad = Instance.new("UIGradient", stroke)
    grad.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0,   Color3.fromRGB(60, 0, 120)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(0, 100, 200)),
        ColorSequenceKeypoint.new(1,   Color3.fromRGB(60, 0, 120)),
    })
    task.spawn(function()
        TweenService:Create(grad, TweenInfo.new(3, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true), {Rotation = 180}):Play()
    end)

    local dragging, dragStart, startPos = false, nil, nil
    btn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = btn.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            btn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                dragging = false
                S.Config[configKey] = {btn.Position.X.Scale, btn.Position.X.Offset, btn.Position.Y.Scale, btn.Position.Y.Offset}
                SaveConfig()
            end
        end
    end)
    return btn, grad
end

local function ToggleFluentUI(visible)
    for _, gui in ipairs(game:GetService("CoreGui"):GetChildren()) do
        if gui:IsA("ScreenGui") then
            local mainFrame = nil
            for _, desc in ipairs(gui:GetDescendants()) do
                if desc:IsA("TextLabel") and desc.Text == "StarCrowHub" then
                    local p = desc.Parent
                    while p and p.Parent ~= gui do p = p.Parent end
                    if p and p:IsA("Frame") then mainFrame = p end
                    break
                end
            end
            if mainFrame then mainFrame.Visible = visible; return end
        end
    end
end

local processedChars = {}

local function removeOutline(character)
    if not character then return end
    local h = character:FindFirstChild("StarCrowOutline")
    if h then h:Destroy() end
end

local function addOutline(character)
    if not character then return end
    local hum = character:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end
    removeOutline(character)
    local h = Instance.new("Highlight")
    h.Name = "StarCrowOutline"
    h.Adornee = character
    h.FillTransparency = 1
    h.OutlineColor = Color3.fromRGB(255, 80, 80)
    h.OutlineTransparency = 0
    h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    h.Parent = character
end

local function removeHeadHitbox(character)
    if not character then return end
    processedChars[character] = nil
    removeOutline(character)
    local head = character:FindFirstChild("Head")
    if not head then return end
    local origSize = head:FindFirstChild("OriginalSize")
    if origSize then head.Size = origSize.Value; origSize:Destroy() end
    local origColor = head:FindFirstChild("OriginalColor")
    if origColor then head.Color = origColor.Value; origColor:Destroy() end
    local meshBackup = head:FindFirstChild("__MeshBackup")
    if meshBackup then
        local realMesh = meshBackup:FindFirstChildOfClass("SpecialMesh") or meshBackup:FindFirstChildOfClass("BlockMesh")
        if realMesh then realMesh:Clone().Parent = head end
        meshBackup:Destroy()
    end
    head.Transparency = 0; head.CanCollide = false; head.Material = Enum.Material.Plastic; head.Massless = false
end

task.spawn(function()
    while true do
        task.wait(1)
        if not S.Config.HeadboxEnabled then
            for _, p in pairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character and processedChars[p.Character] then
                    removeHeadHitbox(p.Character)
                end
            end
            continue
        end
        for _, player in pairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                local character = player.Character
                if character then
                    local humanoid = character:FindFirstChildOfClass("Humanoid")
                    local head = character:FindFirstChild("Head")
                    if humanoid and humanoid.Health > 0 and head then
                        local teamColor = Color3.fromRGB(255,255,255)
                        if player.Team then teamColor = player.Team.TeamColor.Color end
                        local size = S.Config.HeadboxSize
                        local targetSize = Vector3.new(size, size, size)
                        if not head:FindFirstChild("OriginalSize") then
                            local sv = Instance.new("Vector3Value"); sv.Name = "OriginalSize"; sv.Value = head.Size; sv.Parent = head
                        end
                        if not head:FindFirstChild("OriginalColor") then
                            local cv = Instance.new("Color3Value"); cv.Name = "OriginalColor"; cv.Value = head.Color; cv.Parent = head
                        end
                        local meshBackup = head:FindFirstChild("__MeshBackup")
                        if not meshBackup then
                            local existingMesh = head:FindFirstChildOfClass("SpecialMesh") or head:FindFirstChildOfClass("BlockMesh")
                            if existingMesh then
                                local folder = Instance.new("Folder"); folder.Name = "__MeshBackup"; folder.Parent = head
                                existingMesh:Clone().Parent = folder; existingMesh:Destroy()
                            end
                        else
                            local strayMesh = head:FindFirstChildOfClass("SpecialMesh") or head:FindFirstChildOfClass("BlockMesh")
                            if strayMesh then strayMesh:Destroy() end
                        end
                        head.Massless = true; head.Size = targetSize; head.CanCollide = false; head.Transparency = 0.5; head.Material = Enum.Material.ForceField; head.Color = teamColor
                        if not character:FindFirstChild("StarCrowOutline") then addOutline(character) end
                        processedChars[character] = true
                    elseif processedChars[character] then
                        removeHeadHitbox(character)
                    end
                end
            end
        end
    end
end)

local function updateESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char = player.Character
            if char and char:FindFirstChild("Head") and char:FindFirstChild("Humanoid") then
                local hum = char.Humanoid
                local head = char.Head
                local esp = head:FindFirstChild("StarCrowESP")
                local chams = char:FindFirstChild("StarCrowChams")
                if S.Config.ESPEnabled and hum.Health > 0 then
                    if not esp then
                        esp = Instance.new("BillboardGui")
                        esp.Name = "StarCrowESP"; esp.Adornee = head; esp.Size = UDim2.new(0, 350, 0, 30); esp.StudsOffset = Vector3.new(0, 1.2, 0); esp.AlwaysOnTop = true; esp.MaxDistance = math.huge
                        local text = Instance.new("TextLabel")
                        text.Name = "NameText"; text.Parent = esp; text.Size = UDim2.new(1, 0, 1, 0); text.BackgroundTransparency = 1; text.TextScaled = false; text.TextSize = 12; text.Font = Enum.Font.GothamBold; text.TextColor3 = Color3.fromRGB(255, 255, 255); text.TextTransparency = S.Config.ESPTextTransparency; text.TextStrokeTransparency = S.Config.ESPTextTransparency; text.TextStrokeColor3 = Color3.fromRGB(0, 0, 0); text.RichText = true
                        esp.Parent = head
                    end
                    local dist = math.floor((Camera.CFrame.Position - head.Position).Magnitude)
                    local hp = math.floor((hum.Health / hum.MaxHealth) * 100)
                    local teamColor = Color3.fromRGB(255, 255, 255)
                    local tR, tG, tB = 255, 255, 255
                    if player.Team then
                        local tc = player.Team.TeamColor.Color

                    end

local function IsTargetVisible(targetPart, myChar)
    local origin = Camera.CFrame.Position
    local direction = targetPart.Position - origin
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {myChar, Camera}
    params.FilterType = Enum.RaycastFilterType.Exclude
    local result = workspace:Raycast(origin, direction, params)
    return not (result and result.Instance and not result.Instance:IsDescendantOf(targetPart.Parent))
end

local function isFriend(player)
    if not S.Config.FriendCheck then return false end
    local userId = player.UserId
    if FriendCache[userId] ~= nil then return FriendCache[userId] end
    local success, result = pcall(function() return LocalPlayer:IsFriendsWith(userId) end)
    if success then FriendCache[userId] = result; return result end
    return false
end

task.spawn(function()
    while task.wait(0.05) do
        local char = LocalPlayer.Character
        isHoldingGun = char and char:FindFirstChildOfClass("Tool") ~= nil

        if not (S.Config.AimPlayer or S.Config.AimNPC) then currentTarget = nil; continue end
        if not char or not char:FindFirstChild("HumanoidRootPart") then currentTarget = nil; continue end
        if not (S.Config.NoToolCheck or isHoldingGun) then currentTarget = nil; continue end

        local myPos = char.HumanoidRootPart.Position
        local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
        local bestTarget = nil
        local shortest = math.huge

        local function process(obj, isPlayer)
            local hum = obj:FindFirstChildOfClass("Humanoid")
            local part = obj:FindFirstChild(S.Config.AimPart) or obj:FindFirstChild("Head")
            if not hum or hum.Health <= 0 or not part then return end
            if isPlayer then
                local plr = Players:GetPlayerFromCharacter(obj)
                if not plr or plr == LocalPlayer then return end
                if S.Config.TeamCheck and plr.Team == LocalPlayer.Team then return end
                if S.Config.FriendCheck and isFriend(plr) then return end
                if plr.Team and string.match(string.lower(plr.Team.Name), "civilian") then return end
            end
            if S.Config.WallCheck and not IsTargetVisible(part, char) then return end

            local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
            if not onScreen then return end
            if S.Config.AimMode == "FOV" then
                local dist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                if dist < S.Config.FOVRadius and dist < shortest then shortest = dist; bestTarget = part end
            else
                local dist3 = (part.Position - myPos).Magnitude
                if dist3 < shortest then shortest = dist3; bestTarget = part end
            end
        end

        if S.Config.AimPlayer then
            for _, p in ipairs(Players:GetPlayers()) do if p.Character then process(p.Character, true) end end
        end
        if S.Config.AimNPC then
            for i = #NPCCache, 1, -1 do
                local npc = NPCCache[i]
                if npc and npc.Parent and not Players:GetPlayerFromCharacter(npc) then process(npc, false) else table.remove(NPCCache, i) end
            end
        end
        currentTarget = bestTarget
    end
end)

local isNamecallHooked = false
local oldNamecall

function InitSilentAimHook()
    if isNamecallHooked then return end
    isNamecallHooked = true
    oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
        if not (S.Config.AimPlayer or S.Config.AimNPC) then return oldNamecall(self, ...) end
        local method = getnamecallmethod()
        if not checkcaller() and self == workspace and currentTarget and (not S.Config.ClickToAim or S.Config.IsTapping) and (S.Config.NoToolCheck or isHoldingGun) then
            if method == "Raycast" or method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRay" then
                local args = {...}
                local targetPos = currentTarget.Position
                if S.Config.Prediction then
                    local velocity = currentTarget.AssemblyLinearVelocity
                    if velocity then targetPos = targetPos + (velocity * S.Config.PredictionFactor) end
                end
                if method == "Raycast" and typeof(args[2]) == "Vector3" and args[2].Magnitude > 30 then
                    args[2] = (targetPos - args[1]).Unit * args[2].Magnitude
                    return oldNamecall(self, unpack(args))
                elseif (method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRay") and args[1] and args[1].Direction.Magnitude > 30 then
                    args[1] = Ray.new(args[1].Origin, (targetPos - args[1].Origin).Unit * args[1].Direction.Magnitude)
                    return oldNamecall(self, unpack(args))
                end
            end
        end
        return oldNamecall(self, ...)
    end)
end

local function IsBagFull()
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if playerGui then
        for _, desc in ipairs(playerGui:GetDescendants()) do
            if desc:IsA("TextLabel") and desc.Visible and string.find(desc.Text, "Bag:") then
                local cleanText = string.gsub(desc.Text, ",", "")
                local current, max = string.match(cleanText, "%$?(%d+)%s*/%s*%$?(%d+)")
                if current and max then return tonumber(current) >= tonumber(max) end
            end
        end
    end
    return false
end

local function GetPlayerMoneyStat()
    local leaderstats = LocalPlayer:FindFirstChild("leaderstats")
    if leaderstats then
        local bestStat = leaderstats:FindFirstChild("Cash") or leaderstats:FindFirstChild("Money") or leaderstats:FindFirstChild("Coins") or leaderstats:FindFirstChild("Bounty") or leaderstats:FindFirstChild("$$")
        if not bestStat then
            for _, stat in ipairs(leaderstats:GetChildren()) do
                if stat:IsA("IntValue") or stat:IsA("NumberValue") then bestStat = stat; break end
            end
        end
        return bestStat
    end
    return nil
end

local function TriggerRob()
    local robRemote = ReplicatedStorage:WaitForChild("GeneralEvents", 9e9):WaitForChild("Rob", 9e9)
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChild("Humanoid")
    if not root or not hum or hum.Health <= 0 then return end
    for i = 1, 10 do
        if not getgenv().StarCrowAutoBank then break end
        if IsBagFull() then break end
        char = LocalPlayer.Character
        root = char and char:FindFirstChild("HumanoidRootPart")
        hum = char and char:FindFirstChild("Humanoid")
        if not root or not hum or hum.Health <= 0 then break end
        local myPos = root.Position
        local targets = GetRobTargets()
        for _, target in ipairs(targets) do
            if target.part and target.part.Parent and (myPos - target.part.Position).Magnitude <= 30 then
                pcall(function()
                    if target.isSafe then
                        local r = target.obj:FindFirstChild("OpenSafe")
                        if r and r:IsA("RemoteEvent") then r:FireServer("Complete") end
                        robRemote:FireServer("Safe", target.obj)
                    else
                        robRemote:FireServer("Register", target.obj)
                    end
                end)
            end
        end
        task.wait(0.1)
    end
end

local bankActive = false
local waitingForSpawn = false

local function StartBankFarm()
    if bankActive then return end
    bankActive = true
    task.spawn(function()
        local SpawnEvent = ReplicatedStorage:WaitForChild("GeneralEvents", 9e9):WaitForChild("Spawn", 9e9)
        local Banks = {
            {spawn = "Tumbleweed", p1 = Vector3.new(-269.49, 13.35, -172.71), p2 = Vector3.new(-255.19, 13.35, -174.03)},
            {spawn = "Grayridge",  p1 = Vector3.new(1578.65, 121.35, 1676.49), p2 = nil},
            {spawn = "Plainview",  p1 = Vector3.new(-1.89, 51.79, 3023.91), p2 = Vector3.new(5.26, 51.79, 3027.07)},
            {spawn = "StoneCreek", p1 = Vector3.new(873.77, 24.67, 198.60), p2 = nil},
        }
        local bankIndex = 1
        local t0 = 0
        while getgenv().StarCrowAutoBank do
            if waitingForSpawn then task.wait(0.1); continue end
            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChild("Humanoid")
            if not char or not hrp or not hum or hum.Health <= 0 then
                -- die xong reset ngay để không bị kẹt timeout
                t0 = 0
                task.wait(0.5)
                continue
            end
            if IsBagFull() then
                pcall(function() SpawnEvent:InvokeServer("FortCassidy", true, false) end)
                repeat task.wait(0.5) until not IsBagFull() or not getgenv().StarCrowAutoBank
            else
                local bank = Banks[bankIndex]
                waitingForSpawn = true
                local conn
                conn = LocalPlayer.CharacterAdded:Connect(function(newChar)
                    task.wait(0.3)
                    local newHrp = newChar:WaitForChild("HumanoidRootPart", 5)
                    local newHum = newChar:WaitForChild("Humanoid", 5)
                    if newHrp and newHum and newHum.Health > 0 then
                        if bank.p1 and not IsBagFull() then
                            newHrp.CFrame = CFrame.new(bank.p1)
                            newHrp.AssemblyLinearVelocity = Vector3.zero
                            if newHum then newHum:ChangeState(Enum.HumanoidStateType.Jumping) end
                            task.wait(0.5)
                            TriggerRob()
                        end
                        if bank.p2 and not IsBagFull() then
                            newHrp.CFrame = CFrame.new(bank.p2)
                            newHrp.AssemblyLinearVelocity = Vector3.zero
                            if newHum then newHum:ChangeState(Enum.HumanoidStateType.Jumping) end
                            task.wait(0.5)
                            TriggerRob()
                        end
                    end
                    conn:Disconnect()
                    waitingForSpawn = false
                    t0 = 0 -- thoát repeat ngay khi spawn xong
                end)
                pcall(function() SpawnEvent:InvokeServer(bank.spawn, false, false) end)
                t0 = os.clock()
                repeat task.wait(0.1) until not waitingForSpawn or os.clock() - t0 >= 8 or t0 == 0 or not getgenv().StarCrowAutoBank
                if waitingForSpawn then conn:Disconnect(); waitingForSpawn = false end
                bankIndex = bankIndex + 1
                if bankIndex > #Banks then bankIndex = 1 end
            end
            task.wait(0.5)
        end
        waitingForSpawn = false
        bankActive = false
    end)
end

    local itemConfig = { MaxDistance = 1500, FillTransparency = 0.8, OutlineTransparency = 0.2, DefaultColor = Color3.fromRGB(255, 255, 255) }
    local itemHighlights = {}
    local cleanupTick = tick()

    local function getItemColor(item)
        if not item then return itemConfig.DefaultColor end
        if item:IsA("BasePart") then return item.Color end
        if not item:IsA("Model") then return itemConfig.DefaultColor end
        if item.PrimaryPart and item.PrimaryPart.Transparency < 0.8 then
            local primaryName = item.PrimaryPart.Name:lower()
            if not primaryName:find("part") and not primaryName:find("handle") then return item.PrimaryPart.Color end
        end
        local bestPart = nil; local bestScore = 0
        for _, obj in ipairs(item:GetDescendants()) do
            if obj:IsA("BasePart") and obj.Transparency < 0.8 and obj.Size.Magnitude > 0.05 then
                local score = obj.Size.X * obj.Size.Y * obj.Size.Z
                local lowerName = obj.Name:lower()
                if lowerName:find("mesh") then score = score * 2 end
                if not lowerName:find("part") and not lowerName:find("handle") then score = score * 3 end
                if score > bestScore then bestScore = score; bestPart = obj end
            end
        end
        return bestPart and bestPart.Color or itemConfig.DefaultColor
    end

    local function removeItemHighlight(item)
        if itemHighlights[item] then itemHighlights[item]:Destroy(); itemHighlights[item] = nil end
    end

    local function createItemHighlight(item)
        if itemHighlights[item] then return end
        local highlight = Instance.new("Highlight")
        local color = getItemColor(item)
        highlight.Name = "ItemMansionESP"; highlight.FillColor = color; highlight.OutlineColor = color; highlight.FillTransparency = itemConfig.FillTransparency; highlight.OutlineTransparency = itemConfig.OutlineTransparency; highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop; highlight.Enabled = S.Config.ItemMansionESP
        if pcall(function() highlight.Parent = item end) then itemHighlights[item] = highlight end
    end

    local function updateItemHighlights()
        local character = LocalPlayer.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        for item, highlight in pairs(itemHighlights) do
            if not item or not item.Parent then removeItemHighlight(item)
            else
                if root and S.Config.ItemMansionESP then
                    highlight.Enabled = (root.Position - item:GetPivot().Position).Magnitude <= itemConfig.MaxDistance
                else highlight.Enabled = false end
            end
        end
        if tick() > cleanupTick then
            for item in pairs(itemHighlights) do if not item.Parent then itemHighlights[item] = nil end end
            cleanupTick = tick() + 5
        end
    end

    for _, item in ipairs(itemsFolder:GetChildren()) do task.spawn(function() task.wait(0.2); createItemHighlight(item) end) end
    itemsFolder.ChildAdded:Connect(function(item) task.wait(0.3); createItemHighlight(item) end)

local function getClosestDynamite()
    local closest = nil; local dist = math.huge
    if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then return nil end
    for _,plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") and plr.Team and plr.Team.Name ~= "Civilians" then
            if S.Config.DynamiteTeamCheck and LocalPlayer.Team and plr.Team == LocalPlayer.Team then continue end
            local mag = (LocalPlayer.Character.HumanoidRootPart.Position - plr.Character.HumanoidRootPart.Position).Magnitude
            if mag < dist then dist = mag; closest = plr end
        end
    end
    return closest
end

workspace.ChildAdded:Connect(function(v)
    if v.Name == "DynamitePart" then
        task.wait()
        if not S.Config.DynamiteTargetEnabled then return end
        local target = getClosestDynamite()
        if not target then return end
        local hrp = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        v.CanCollide = false
        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge); bv.Parent = v
        local bg = Instance.new("BodyGyro")
        bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge); bg.P = 999999; bg.Parent = v

        local conn
        conn = RunService.Heartbeat:Connect(function()
            if not S.Config.DynamiteTargetEnabled then
                if conn then conn:Disconnect() end
                if bv then bv:Destroy() end
                if bg then bg:Destroy() end
                return
            end
            if v and v.Parent and hrp and hrp.Parent then
                local pos = hrp.Position + Vector3.new(0,3,0)
                bv.Velocity = (pos - v.Position) * 25
                bg.CFrame = CFrame.new(v.Position, hrp.Position)
            else
                if conn then conn:Disconnect() end
            end
        end)
    end
end)
