-- AUTO SAVE
local CONFIG_FOLDER = "StarCrowHub"
local CONFIG_FILE   = CONFIG_FOLDER .. "/GunArmoryTycoonConfig.json"

local function EnsureFolder()
    if not isfolder(CONFIG_FOLDER) then
        makefolder(CONFIG_FOLDER)
    end
end

local function SaveConfig()
    pcall(function()
        EnsureFolder()
        local data = {
            Features        = Features,
            Values          = Values,
            SilentAimSettings = {
                Enabled         = SilentAimSettings.Enabled,
                ShowFOV         = SilentAimSettings.ShowFOV,
                FOVRadius       = SilentAimSettings.FOVRadius,
                WallbangEnabled = SilentAimSettings.WallbangEnabled,
            },
            Config = Config,
        }
        writefile(CONFIG_FILE, HttpService:JSONEncode(data))
    end)
end

local function LoadConfig()
    pcall(function()
        EnsureFolder()
        if isfile(CONFIG_FILE) then
            local decoded = HttpService:JSONDecode(readfile(CONFIG_FILE))
            if type(decoded) ~= "table" then return end

            if type(decoded.Features) == "table" then
                for k, v in pairs(decoded.Features) do
                    if Features[k] ~= nil then Features[k] = v end
                end
            end
            if type(decoded.Values) == "table" then
                for k, v in pairs(decoded.Values) do
                    if Values[k] ~= nil then Values[k] = v end
                end
            end
            if type(decoded.SilentAimSettings) == "table" then
                for k, v in pairs(decoded.SilentAimSettings) do
                    if SilentAimSettings[k] ~= nil then SilentAimSettings[k] = v end
                end
            end
            if type(decoded.Config) == "table" then
                for k, v in pairs(decoded.Config) do
                    if Config[k] ~= nil then Config[k] = v end
                end
            end
        end
    end)
end

-- ESP LOGIC
local espObjects = {}

local function updateESP()
    if not Config.ESPEnabled then
        for player, data in pairs(espObjects) do
            if data.bill  then data.bill:Destroy()  end
            if data.chams then data.chams:Destroy() end
        end
        espObjects = {}
        return
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end

        local char   = player.Character
        local root   = char and char:FindFirstChild("HumanoidRootPart")
        local hum    = char and char:FindFirstChildOfClass("Humanoid")
        local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

        if char and root and hum and hum.Health > 0 then
            if not espObjects[player] then
                local bill = Instance.new("BillboardGui")
                bill.Name = "StarCrowESP"
                bill.Size = UDim2.new(0, 200, 0, 50)
                bill.StudsOffset = Vector3.new(0, 3, 0)
                bill.AlwaysOnTop = true
                bill.Parent = root

                local nameLabel = Instance.new("TextLabel")
                nameLabel.Name = "NameText"
                nameLabel.Size = UDim2.new(1, 0, 1, 0)
                nameLabel.BackgroundTransparency = 1
                nameLabel.TextStrokeTransparency = 0
                nameLabel.TextStrokeColor3 = Color3.new(0, 0, 0)
                nameLabel.RichText = true
                nameLabel.Font = Enum.Font.GothamBold
                nameLabel.TextSize = 13
                nameLabel.Parent = bill

                local chams = Instance.new("Highlight")
                chams.Name = "StarCrowChams"
                chams.FillTransparency = 0.7
                chams.OutlineTransparency = 0
                chams.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                chams.Parent = char

                espObjects[player] = { bill = bill, nameText = nameLabel, chams = chams }
            end

            local data    = espObjects[player]
            local hp      = math.floor((hum.Health / math.max(hum.MaxHealth, 1)) * 100)
            local dist    = myRoot and math.floor((root.Position - myRoot.Position).Magnitude) or 0
            local hpColor = hp > 70 and "rgb(0,255,127)" or hp > 30 and "rgb(255,200,0)" or "rgb(255,50,50)"

            local tR, tG, tB = 255, 255, 255
            if player.Team then
                local tc = player.Team.TeamColor.Color
                tR, tG, tB = math.floor(tc.R*255), math.floor(tc.G*255), math.floor(tc.B*255)
                data.chams.FillColor    = tc
                data.chams.OutlineColor = tc
            end

            data.nameText.Text = string.format(
                '<font color="rgb(%d,%d,%d)">%s</font> | %dm | <font color="%s">%d%%</font>',
                tR, tG, tB, player.DisplayName, dist, hpColor, hp
            )

            if data.bill.Parent  ~= root then data.bill.Parent  = root end
            if data.chams.Parent ~= char then data.chams.Parent = char end
        else
            if espObjects[player] then
                if espObjects[player].bill  then espObjects[player].bill:Destroy()  end
                if espObjects[player].chams then espObjects[player].chams:Destroy() end
                espObjects[player] = nil
            end
        end
    end

    for player in pairs(espObjects) do
        if not player or not player.Parent then
            if espObjects[player].bill  then espObjects[player].bill:Destroy()  end
            if espObjects[player].chams then espObjects[player].chams:Destroy() end
            espObjects[player] = nil
        end
    end
end

        if espObjects[player].bill  then espObjects[player].bill:Destroy()  end
        if espObjects[player].chams then espObjects[player].chams:Destroy() end
        espObjects[player] = nil
    end
end)

-- RENDERSTEPPED
local currentTarget = nil
local isHoldingGun  = false

RunService.RenderStepped:Connect(function()
    Camera = workspace.CurrentCamera
    local char = LocalPlayer.Character
    isHoldingGun = char and char:FindFirstChildOfClass("Tool") ~= nil

    local target = nil
    if isHoldingGun and SilentAimSettings.Enabled then
        local shortestDist = SilentAimSettings.FOVRadius
        local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local part = player.Character:FindFirstChild(SilentAimSettings.AimPart)
                local hum  = player.Character:FindFirstChildOfClass("Humanoid")
                if part and hum and hum.Health > 0 then
                    local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
                    if onScreen then
                        local dist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                        if dist < shortestDist then
                            target = part
                            shortestDist = dist
                        end
                    end
                end
            end
        end
    end
    currentTarget = target

    fovCircle.Visible = SilentAimSettings.Enabled and SilentAimSettings.ShowFOV

    updateESP()
end)

-- WALLBANG HOOK
local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()

    if SilentAimSettings.Enabled and not checkcaller() and currentTarget and isHoldingGun and self == workspace then
        local args = {...}

        if method == "Raycast" then
            local origin    = args[1]
            local direction = args[2]
            if typeof(origin) == "Vector3" and typeof(direction) == "Vector3" and direction.Magnitude > 30 then
                args[2] = (currentTarget.Position - origin).Unit * 5000
                if SilentAimSettings.WallbangEnabled then
                    local newParams = RaycastParams.new()
                    newParams.FilterType = Enum.RaycastFilterType.Include
                    newParams.FilterDescendantsInstances = {currentTarget.Parent}
                    args[3] = newParams
                end
                return oldNamecall(self, unpack(args))
            end

        elseif method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRay" then
            local ray = args[1]
            if typeof(ray) == "Ray" and ray.Direction.Magnitude > 30 then
                args[1] = Ray.new(ray.Origin, (currentTarget.Position - ray.Origin).Unit * 5000)
                return oldNamecall(self, unpack(args))
            end
        end
    end

    return oldNamecall(self, ...)
end)

-- ACS MOD LOGIC
local function ModWeapon(tool)
    if not tool:IsA("Tool") then return end
    local mod = tool:FindFirstChild("ACS_Settings")
    if not mod then return end
    local ok, t = pcall(require, mod)
    if not (ok and type(t) == "table") then return end

    if Features.NoRecoil then
        t.camRecoil = {camRecoilUp={0,0}, camRecoilTilt={0,0}, camRecoilLeft={0,0}, camRecoilRight={0,0}}
        t.gunRecoil = {gunRecoilUp={0,0}, gunRecoilTilt={0,0}, gunRecoilLeft={0,0}, gunRecoilRight={0,0}}
        t.AimRecoilReduction, t.MinRecoilPower, t.MaxRecoilPower = 0, 0, 0
        t.RecoilPowerStepAmount, t.MinSpread, t.MaxSpread = 0, 0, 0
        t.AimInaccuracyDecrease, t.AimInaccuracyStepAmount, t.AimSpreadReduction, t.WalkMult = 0, 0, 0, 0
    end
    if Features.SpreadLock then
        t.MinSpread, t.MaxSpread = Values.MinSpreadVal, Values.MaxSpreadVal
        t.AimInaccuracyDecrease, t.AimInaccuracyStepAmount = 0, 0
    end
    if Features.RapidFire    then t.ShootRate = Values.ShootRate end
    if Features.FullAuto     then
        t.FireModes = {ChangeFiremode=true, Semi=true, Burst=true, Auto=true}
        t.ShootType = 3
    end
    t.IgnoreProtection = Features.ArmorPiercing
    if Features.InfAmmo      then
        t.Ammo, t.AmmoInGun, t.StoredAmmo, t.MaxStoredAmmo = Values.AmmoAmount, Values.AmmoAmount, Values.AmmoAmount, Values.AmmoAmount
    end
    if Features.NoBulletDrop then t.BulletDrop = 0 end
    if Features.Rainbow      then t.RainbowMode, t.Tracer = true, true else t.RainbowMode = false end
    if Features.Tracer and not Features.Rainbow then t.Tracer = true end
    if Features.NoJam        then t.CanBreak, t.Jammed = false, false end
    if Features.NoFlash      then t.FlashChance = 0 end
end

local function ApplyAll()
    for _, item in ipairs(LocalPlayer.Backpack:GetChildren()) do ModWeapon(item) end
    if LocalPlayer.Character then
        for _, item in ipairs(LocalPlayer.Character:GetChildren()) do ModWeapon(item) end
    end
end
