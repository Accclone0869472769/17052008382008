local INVIS_CONFIG = {
    INVISIBILITY_POSITION = Vector3.new(-25.95,84,3537.55)
}

local InvisState = {
    isInvisible = false
}

local function setCharacterTransparency(character, transparency)
    if not character then return end

    for _, d in character:GetDescendants() do
        if d:FindFirstAncestorOfClass("Accessory") then
            continue
        end

        if d:IsA("BasePart") then
            if d.Name == "HumanoidRootPart" then
                d.Transparency = 1
            elseif d.Name == "Head" then
                d.Transparency = 0
            else
                d.Transparency = transparency
            end

        elseif d:IsA("Decal") then
            if d.Name == "face" and d.Parent and d.Parent.Name == "Head" then
                d.Transparency = 0
            else
                d.Transparency = transparency
            end
        end
    end
end

local function getHumanoidRootPart()
    local character = LocalPlayer.Character
    if not character then
        return nil
    end

    return character:FindFirstChild("HumanoidRootPart")
end

local function toggleInvisibility(state)
    if not LocalPlayer.Character then
        return
    end
    
    if InvisState.isInvisible == state then return end
    InvisState.isInvisible = state

    local cam = workspace.CurrentCamera

    if InvisState.isInvisible then
        local hrp = getHumanoidRootPart()
        if not hrp then
            return
        end

        local savedCFrame = hrp.CFrame
        local savedCamCFrame = cam.CFrame

        cam.CameraType = Enum.CameraType.Scriptable
        cam.CFrame = savedCamCFrame

        LocalPlayer.Character:MoveTo(INVIS_CONFIG.INVISIBILITY_POSITION)

        task.wait(0.1)

        local seat = Instance.new("Seat")
        seat.Name = "invischair"
        seat.Anchored = false
        seat.CanCollide = false
        seat.Transparency = 1
        seat.Size = Vector3.new(2,0.5,2)
        seat.Position = INVIS_CONFIG.INVISIBILITY_POSITION
        seat.Parent = workspace

        local weld = Instance.new("Weld")
        weld.Part0 = seat
        weld.Part1 = hrp
        weld.Parent = seat

        task.wait()

        task.defer(function()
            if seat and seat.Parent then
                seat.CFrame = savedCFrame
            end
        end)

        cam.CameraType = Enum.CameraType.Custom

        local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            cam.CameraSubject = humanoid
        end

        setCharacterTransparency(LocalPlayer.Character, 0.5)
        hrp.Transparency = 1

    else
        local chair = workspace:FindFirstChild("invischair")
        if chair then
            chair:Destroy()
        end

        if LocalPlayer.Character then
            setCharacterTransparency(LocalPlayer.Character, 0)
            local hrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Transparency = 1
            end

            local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                cam.CameraSubject = humanoid
            end
        end
    end
end

local LoopBring = {
    Enabled = false,
    Targets = {},
    Connection = nil
}

local function StartLoopBring()
    if LoopBring.Connection then
        LoopBring.Connection:Disconnect()
        LoopBring.Connection = nil
    end

    LoopBring.Enabled = true

    LoopBring.Connection = RunService.Heartbeat:Connect(function()
        if not LoopBring.Enabled then
            return
        end

        local myChar = LocalPlayer.Character
        local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")

        if not myHRP then
            return
        end

        for targetName, isSelected in pairs(LoopBring.Targets) do
            if isSelected then
                local targetPlayer = Players:FindFirstChild(targetName)
                if targetPlayer then
                    local targetChar = targetPlayer.Character
                    local targetHRP = targetChar and targetChar:FindFirstChild("HumanoidRootPart")

                    if targetHRP then
                        targetHRP.CFrame = myHRP.CFrame * CFrame.new(0, 0, -3)
                    end
                end
            end
        end
    end)
end

local function StopLoopBring()
    LoopBring.Enabled = false

    if LoopBring.Connection then
        LoopBring.Connection:Disconnect()
        LoopBring.Connection = nil
    end
end

local SAVE_FOLDER = "StarCrowHub"
local SAVE_FILE = SAVE_FOLDER .. "/GunTycoonConfig.json"

local function SaveConfig()
    pcall(function()
        if not isfolder(SAVE_FOLDER) then makefolder(SAVE_FOLDER) end
        writefile(SAVE_FILE, HttpService:JSONEncode(Config))
    end)
end

local function LoadConfig()
    pcall(function()
        if isfile(SAVE_FILE) then
            local raw = readfile(SAVE_FILE)
            local data = HttpService:JSONDecode(raw)
            for k, v in pairs(data) do
                if Config[k] ~= nil then
                    Config[k] = v
                end
            end
        end
    end)
end

local currentTarget = nil
local isHoldingGun = false

local FriendCache = {}
local function isFriend(player)
    if FriendCache[player.UserId] ~= nil then return FriendCache[player.UserId] end
    local s, r = pcall(function() return LocalPlayer:IsFriendsWith(player.UserId) end)
    if s then FriendCache[player.UserId] = r return r end
    return false
end

task.spawn(function()
    while task.wait(0.1) do
        if not Config.SilentAim and not Config.MagicBullets then
            currentTarget = nil
            continue
        end

        local char = LocalPlayer.Character
        isHoldingGun = char and char:FindFirstChildOfClass("Tool") ~= nil
        
        local target = nil
        if isHoldingGun then
            local shortestDist = Config.FOV
            local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    if Config.TeamCheck and player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then continue end
                    if Config.FriendCheck and isFriend(player) then continue end

                    local part = player.Character:FindFirstChild(Config.AimPart)
                    local hum = player.Character:FindFirstChildOfClass("Humanoid")
                    
                    if part and hum and hum.Health > 0 then
                        local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
                        if onScreen then
                            local dist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                            if dist < shortestDist then
                                target = part
                                shortestDist = dist
                            end
                        end
                    end
                end
            end
        end
        currentTarget = target
    end
end)

local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    if not Config.SilentAim and not Config.MagicBullets then return oldNamecall(self, ...) end 

    local method = getnamecallmethod()
    
    if not checkcaller() and currentTarget and isHoldingGun and self == workspace then
        local args = {...}
        
        if method == "Raycast" then
            local origin = args[1]
            local direction = args[2]
            
            if typeof(origin) == "Vector3" and typeof(direction) == "Vector3" and direction.Magnitude > 5 then
                local targetPos = currentTarget.Position
                if Config.MagicBullets then
                    local dirToTarget = (targetPos - Camera.CFrame.Position).Unit
                    args[1] = targetPos - (dirToTarget * 1.5)
                    args[2] = dirToTarget * 100
                else
                    args[2] = (targetPos - origin).Unit * direction.Magnitude
                end
                return oldNamecall(self, unpack(args))
            end
            
        elseif method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRay" then
            local ray = args[1]
            if typeof(ray) == "Ray" and ray.Direction.Magnitude > 5 then
                local targetPos = currentTarget.Position
                if Config.MagicBullets then
                    local dirToTarget = (targetPos - Camera.CFrame.Position).Unit
                    local newOrigin = targetPos - (dirToTarget * 1.5)
                    args[1] = Ray.new(newOrigin, dirToTarget * 100)
                else
                    args[1] = Ray.new(ray.Origin, (targetPos - ray.Origin).Unit * ray.Direction.Magnitude)
                end
                return oldNamecall(self, unpack(args))
            end
        end
    end
    
    return oldNamecall(self, ...)
end)

local GunCache = {}
local OriginalStats = {}

local function ScanGuns()
    for _, v in pairs(getgc(true)) do
        if type(v) == "table" and rawget(v, "ModuleName") and rawget(v, "FireRate") then
            if not OriginalStats[v] then
                OriginalStats[v] = {
                    LimitedAmmoEnabled = rawget(v, "LimitedAmmoEnabled"),
                    AmmoCost = rawget(v, "AmmoCost"),
                    Ammo = rawget(v, "Ammo"), 
                    MaxAmmo = rawget(v, "MaxAmmo"),
                    ReloadTime = rawget(v, "ReloadTime"),
                    Recoil = rawget(v, "Recoil"),
                    Spread = rawget(v, "Spread"),
                    CameraRecoilingEnabled = rawget(v, "CameraRecoilingEnabled"),
                    Auto = rawget(v, "Auto"),
                    FireRate = rawget(v, "FireRate"),
                    ShotgunEnabled = rawget(v, "ShotgunEnabled"),
                    BulletPerShot = rawget(v, "BulletPerShot"),
                    ShotgunPattern = rawget(v, "ShotgunPattern"),
                    BaseDamage = rawget(v, "BaseDamage")
                }
                table.insert(GunCache, v)
            end
        end
    end
end

local function UpdateGuns()
    for _, v in ipairs(GunCache) do
        pcall(function()
            if setreadonly then setreadonly(v, false) end

            v.LimitedAmmoEnabled = not Config.InfAmmo and OriginalStats[v].LimitedAmmoEnabled or false
            v.AmmoCost = Config.InfAmmo and 0 or OriginalStats[v].AmmoCost
            v.Ammo = Config.InfAmmo and 999999 or OriginalStats[v].Ammo
            v.MaxAmmo = Config.InfAmmo and 999999 or OriginalStats[v].MaxAmmo
            v.ReloadTime = Config.InfAmmo and 0 or OriginalStats[v].ReloadTime

            v.Recoil = Config.NoRecoil and 0 or OriginalStats[v].Recoil
            v.Spread = Config.NoRecoil and 0 or OriginalStats[v].Spread
            v.CameraRecoilingEnabled = not Config.NoRecoil and OriginalStats[v].CameraRecoilingEnabled or false

            v.Auto = Config.FastFire and true or OriginalStats[v].Auto
            v.FireRate = Config.FastFire and Config.FireRate or OriginalStats[v].FireRate

            v.ShotgunEnabled = Config.Multishot and true or OriginalStats[v].ShotgunEnabled
            v.BulletPerShot = Config.Multishot and Config.BulletPerShot or OriginalStats[v].BulletPerShot
            v.ShotgunPattern = not Config.Multishot and OriginalStats[v].ShotgunPattern or false

            if setreadonly then setreadonly(v, true) end
        end)
    end
end

local function ApplyModsInstant()
    UpdateGuns()
end

ScanGuns()

local function updateESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char = player.Character
            if char and char:FindFirstChild("Head") and char:FindFirstChild("Humanoid") then
                local hum = char.Humanoid
                local head = char.Head
                local esp = head:FindFirstChild("StarCrowESP")
                local chams = char:FindFirstChild("StarCrowChams")

                if Config.ESPEnabled and hum.Health > 0 then
                    if not esp then
                        esp = Instance.new("BillboardGui")
                        esp.Name = "StarCrowESP"
                        esp.Adornee = head
                        esp.Size = UDim2.new(0, 350, 0, 30)
                        esp.StudsOffset = Vector3.new(0, 1.2, 0)
                        esp.AlwaysOnTop = true
                        esp.MaxDistance = math.huge
                        local text = Instance.new("TextLabel")
                        text.Name = "NameText"
                        text.Parent = esp
                        text.Size = UDim2.new(1, 0, 1, 0)
                        text.BackgroundTransparency = 1
                        text.TextScaled = false
                        text.TextSize = 12
                        text.Font = Enum.Font.GothamBold
                        text.TextColor3 = Color3.fromRGB(255, 255, 255)
                        text.TextStrokeTransparency = 0
                        text.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                        text.RichText = true
                        esp.Parent = head
                    end
                    local dist = math.floor((Camera.CFrame.Position - head.Position).Magnitude)
                    local hp = math.floor((hum.Health / hum.MaxHealth) * 100)
                    local teamColor = Color3.fromRGB(255, 255, 255)
                    local tR, tG, tB = 255, 255, 255
                    if player.Team then
                        local tc = player.Team.TeamColor.Color
                        teamColor = tc
                        tR, tG, tB = math.floor(tc.R*255), math.floor(tc.G*255), math.floor(tc.B*255)
                    end
                    local hpColor = hp > 70 and "rgb(0, 255, 127)" or hp > 30 and "rgb(255, 200, 0)" or "rgb(255, 50, 50)"
                    esp.NameText.Text = string.format('<font color="rgb(%d,%d,%d)">%s</font> | %dm | <font color="%s">%d%%</font>', tR, tG, tB, player.DisplayName, dist, hpColor, hp)
                    
                    if not chams then
                        chams = Instance.new("Highlight")
                        chams.Name = "StarCrowChams"
                        chams.Parent = char
                    end
                    chams.FillColor = teamColor
                    chams.OutlineColor = teamColor
                    chams.FillTransparency = 0.7
                    chams.OutlineTransparency = 0
                    chams.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                else
                    if esp then esp:Destroy() end
                    if chams then chams:Destroy() end
                end
            end
        end
    end
end

    Callback = function(v) Config.NoclipEnabled = v; SaveConfig() end
})

Tabs.Player:AddToggle("LoopSpeed", {
    Title = "Enable WalkSpeed", Default = Config.LoopSpeedEnabled,
    Callback = function(v) Config.LoopSpeedEnabled = v; SaveConfig() end
})

Tabs.Player:AddSlider("WalkSpeed", {
    Title = "Speed Amount", Default = Config.WalkSpeed, Min = 16, Max = 250, Rounding = 0,
    Callback = function(v) Config.WalkSpeed = v; SaveConfig() end
})

Tabs.Player:AddToggle("InfJump", {
    Title = "Infinite Jump", Default = Config.InfJumpEnabled,
    Callback = function(v) Config.InfJumpEnabled = v; SaveConfig() end
})

Tabs.Settings:AddKeybind("ToggleUIKeybind", {
    Title = "Minimize UI Keybind",
    Mode = "Toggle",
    Default = "LeftControl",
    ChangedCallback = function(NewKey)
        Window.MinimizeKey = NewKey
    end
})

Window:SelectTab(1)
