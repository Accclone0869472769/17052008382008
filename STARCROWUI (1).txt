-- Drag logic (toggleBtn)
local _tbDragging, _tbMoved, _tbIsTouch = false, false, false
local _tbDragInput, _tbDragStart, _tbDragPos

toggleBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        _tbDragging = true; _tbMoved = false; _tbIsTouch = false
        _tbDragStart = input.Position; _tbDragPos = toggleBtn.Position
    elseif input.UserInputType == Enum.UserInputType.Touch then
        _tbDragging = true; _tbMoved = false; _tbIsTouch = true
        _tbDragInput = input; _tbDragStart = input.Position; _tbDragPos = toggleBtn.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                _tbDragging = false; _tbDragInput = nil
            end
        end)
    end
end)
toggleBtn.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        _tbDragging = false
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if not _tbDragging then return end
    local delta
    if not _tbIsTouch and input.UserInputType == Enum.UserInputType.MouseMovement then
        delta = input.Position - _tbDragStart
    elseif _tbIsTouch and input == _tbDragInput then
        delta = input.Position - _tbDragStart
    end
    if delta then
        if delta.Magnitude > 6 then _tbMoved = true end
        toggleBtn.Position = UDim2.new(
            _tbDragPos.X.Scale, _tbDragPos.X.Offset + delta.X,
            _tbDragPos.Y.Scale, _tbDragPos.Y.Offset + delta.Y
        )
    end
end)

-- ──────────────────────────────────────────────────────────
--  TAB SWITCHING
-- ──────────────────────────────────────────────────────────

local function switchTab(idx)
    for i, p in ipairs(panels) do p.Visible = (i == idx) end
    for i, btn in ipairs(tabBtns) do
        local active = (i == idx)
        btn.BackgroundTransparency = active and 0 or 0.35
        btn.BackgroundColor3 = active
            and Color3.fromRGB(90,20,160)
            or  Color3.fromRGB(35,15,65)

        local bG = btn:FindFirstChildOfClass("UIGradient")
        if bG then
            bG.Color = active
                and ColorSequence.new({
                    ColorSequenceKeypoint.new(0, Color3.fromRGB(140,40,255)),
                    ColorSequenceKeypoint.new(1, Color3.fromRGB(60,10,140)),
                })
                or ColorSequence.new({
                    ColorSequenceKeypoint.new(0, Color3.fromRGB(55,15,100)),
                    ColorSequenceKeypoint.new(1, Color3.fromRGB(15,20,60)),
                })
        end

        local bar = btn:FindFirstChild("AccentBar")
        if not bar then
            bar = Instance.new("Frame")
            bar.Name = "AccentBar"
            bar.Size = UDim2.new(0,3,0.65,0)
            bar.Position = UDim2.new(0,1,0.175,0)
            bar.BorderSizePixel = 0
            bar.ZIndex = btn.ZIndex+1
            _corner(2, bar)
            local barG = Instance.new("UIGradient")
            barG.Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0, Color3.fromRGB(170,70,255)),
                ColorSequenceKeypoint.new(1, Color3.fromRGB(0,200,255)),
            })
            barG.Rotation = 90; barG.Parent = bar
            bar.Parent = btn
        end
        bar.BackgroundColor3 = Color3.fromRGB(255,255,255)
        bar.Visible = active

        local ic = btn:FindFirstChild("TabIcon")
        if ic then ic.ImageColor3 = active and Color3.fromRGB(220,160,255) or Color3.fromRGB(170,140,210) end
        local lb = btn:FindFirstChild("TabLabel")
        if lb then lb.TextColor3 = active and Color3.fromRGB(255,210,255) or Color3.fromRGB(180,150,220) end
    end
end

local function closeMenu()
    local tw = TweenService:Create(mainFrame,
        TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
        {Position = UDim2.new(0.5,-160,0.5,-175), BackgroundTransparency = 0.95}
    )
    tw:Play()
    tw.Completed:Connect(function()
        mainFrame.Visible = false
        mainFrame.BackgroundTransparency = 0.7
        toggleBtn.Size               = UDim2.new(0,170,0,32)
        toggleBtn.BackgroundTransparency = 0.9
        toggleBtn.Visible            = true
        _tween(toggleBtn, TweenInfo.new(0.2, Enum.EasingStyle.Back),

-- Float button click
toggleBtn.MouseButton1Click:Connect(function()
    if _tbMoved then _tbMoved = false; return end

    -- particle burst
    local colors = {
        Color3.fromRGB(160,60,255), Color3.fromRGB(0,180,255),
        Color3.fromRGB(220,100,255), Color3.fromRGB(80,220,255),
    }
    for i = 1, 12 do
        local angle = (i/12)*math.pi*2
        local dist  = 38 + math.random(0,24)
        local sz    = math.random(3,7)
        local p     = Instance.new("Frame")
        p.Size = UDim2.new(0,sz,0,sz)
        p.AnchorPoint = Vector2.new(0.5,0.5)
        p.Position = UDim2.new(0.5,0,0.5,0)
        p.BackgroundColor3 = colors[(i%#colors)+1]
        p.BackgroundTransparency = 0
        p.BorderSizePixel = 0
        p.ZIndex = toggleBtn.ZIndex+8
        p.Parent = toggleBtn
        _corner(999, p)
        _tween(p, TweenInfo.new(0.45, Enum.EasingStyle.Quad), {
            Position = UDim2.new(0.5, math.cos(angle)*dist, 0.5, math.sin(angle)*dist),
            BackgroundTransparency = 1,
            Size = UDim2.new(0,sz*0.4,0,sz*0.4),
        })
        game:GetService("Debris"):AddItem(p, 0.5)
    end

    _tween(toggleBtn, TweenInfo.new(0.08, Enum.EasingStyle.Quad), {Size = UDim2.new(0,210,0,32)})
    task.delay(0.08, function()
        _tween(toggleBtn, TweenInfo.new(0.22, Enum.EasingStyle.Elastic), {Size = UDim2.new(0,190,0,38)})
    end)

    toggleMenu()
end)

-- ─── AddToggle ───────────────────────────────────────────
-- GUI:AddToggle(tabIndex, "Label", callback)
-- callback(state: bool) được gọi mỗi khi bật/tắt
-- Trả về { SetState(bool), GetState() -> bool }
function API:AddToggle(tabIndex, labelText, callback)
    local parent = panels[tabIndex]
    if not parent then return end
    callback = callback or function() end

    local btn = Instance.new("TextButton")
    btn.Size                   = UDim2.new(1,0,0,36)
    btn.BackgroundColor3       = Color3.fromRGB(38,38,38)
    btn.BackgroundTransparency = 0.35
    btn.BorderSizePixel        = 0
    btn.Text                   = ""
    btn.AutoButtonColor        = false
    btn.Parent                 = parent
    _corner(8, btn)

    local bgG = Instance.new("UIGradient")
    bgG.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(255,255,255)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(200,200,255)),
    })
    bgG.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.1),
        NumberSequenceKeypoint.new(1, 0.25),
    })
    bgG.Rotation = 90; bgG.Parent = btn

    local nameLbl = Instance.new("TextLabel")
    nameLbl.Size              = UDim2.new(1,-58,1,0)
    nameLbl.Position          = UDim2.new(0,12,0,0)
    nameLbl.BackgroundTransparency = 1
    nameLbl.Text              = labelText
    nameLbl.TextColor3        = Color3.fromRGB(255,255,255)
    nameLbl.Font              = Enum.Font.GothamSemibold
    nameLbl.TextSize          = 13
    nameLbl.TextXAlignment    = Enum.TextXAlignment.Left
    nameLbl.TextYAlignment    = Enum.TextYAlignment.Center
    nameLbl.ZIndex            = btn.ZIndex+2
    nameLbl.Parent            = btn

    local track = Instance.new("Frame")
    track.Size                   = UDim2.new(0,42,0,22)
    track.Position               = UDim2.new(1,-50,0.5,-11)
    track.BackgroundColor3       = Color3.fromRGB(38,38,38)
    track.BackgroundTransparency = 0.35
    track.BorderSizePixel        = 0
    track.ZIndex                 = btn.ZIndex+2
    track.Parent                 = btn
    _corner(999, track)

    local knob = Instance.new("Frame")
    knob.Size             = UDim2.new(0,18,0,18)
    knob.Position         = UDim2.new(0,2,0.5,-9)
    knob.BackgroundColor3 = Color3.fromRGB(190,190,190)
    knob.BorderSizePixel  = 0
    knob.ZIndex           = btn.ZIndex+3
    knob.Parent           = track
    _corner(999, knob)

    local GREEN = Color3.fromRGB(50,200,120)
    local GREY  = Color3.fromRGB(38,38,38)
    local isOn  = false

    local function sync()
        _tween(track, TweenInfo.new(0.15, Enum.EasingStyle.Quad), {
            BackgroundColor3       = isOn and GREEN or GREY,
            BackgroundTransparency = isOn and 0 or 0.35,
        })
        _tween(knob, TweenInfo.new(0.15, Enum.EasingStyle.Quad), {
            Position         = isOn and UDim2.new(0,22,0.5,-9) or UDim2.new(0,2,0.5,-9),
            BackgroundColor3 = isOn and Color3.fromRGB(255,255,255) or Color3.fromRGB(190,190,190),
        })
    end

    btn.MouseButton1Click:Connect(function()
        _click()
        isOn = not isOn
        sync()
        callback(isOn)
    end)

    return {
        SetState = function(_, state)
            isOn = not not state
            sync()
            callback(isOn)
        end,
        GetState = function() return isOn end,
    }
end
